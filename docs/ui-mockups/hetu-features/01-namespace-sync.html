Generated policy (preview)
apiVersion: kyverno.io/v1
kind: ClusterPolicy
metadata:
  name: hetu-sync-brevo-smtp
spec:
  generateExisting: true
  rules:
  - name: copy-brevo-smtp
    match:
      any:
      - resources:
          kinds: [Namespace]
          selector:
            matchLabels: {hetu/sync: "true"}
    generate:
      apiVersion: v1
      kind: Secret
      name: brevo-smtp
      namespace: "{{request.object.metadata.name}}"
      synchronize: true
      clone:
        namespace: utilities
        name: brevo-smtp
